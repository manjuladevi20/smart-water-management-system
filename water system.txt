# Import libraries
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error
import matplotlib.pyplot as plt

# ----------------------------
# 1. Simulate Sensor Data
# ----------------------------
np.random.seed(42)
n_samples = 200
data = pd.DataFrame({
    'hour': np.tile(np.arange(0, 24), n_samples // 24),
    'day_of_week': np.random.randint(0, 7, size=n_samples),
    'temperature': np.random.normal(25, 5, size=n_samples),  # °C
    'humidity': np.random.normal(60, 10, size=n_samples),     # %
    'flow_rate': np.random.normal(120, 30, size=n_samples)    # Liters/hour
})

# Generate synthetic water demand values
data['water_demand'] = (data['flow_rate'] * 0.8 +
                        data['temperature'] * 0.5 -
                        data['humidity'] * 0.2 +
                        data['hour'] * 2 +
                        np.random.normal(0, 10, size=n_samples))

# ----------------------------
# 2. Train ML Model
# ----------------------------
X = data[['hour', 'day_of_week', 'temperature', 'humidity', 'flow_rate']]
y = data['water_demand']

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)
model = RandomForestRegressor(n_estimators=100)
model.fit(X_train, y_train)

# ----------------------------
# 3. Predict + Detect Anomalies
# ----------------------------
predictions = model.predict(X_test)
errors = abs(predictions - y_test)

# Set anomaly threshold
threshold = 30  # Tune as needed
anomalies = errors > threshold

# Print basic results
print("Mean Squared Error:", mean_squared_error(y_test, predictions))
print("Detected anomalies:", sum(anomalies))

# ----------------------------
# 4. Visualize
# ----------------------------
plt.figure(figsize=(12, 6))
plt.plot(np.arange(len(y_test)), y_test.values, label='Actual Demand', marker='o')
plt.plot(np.arange(len(y_test)), predictions, label='Predicted Demand', linestyle='--', marker='x')
plt.scatter(np.where(anomalies)[0], y_test.values[anomalies], color='red', label='Anomaly', zorder=5)
plt.xlabel('Sample Index')
plt.ylabel('Water Demand (Liters/hour)')
plt.title('Smart Water Demand Prediction & Anomaly Detection')
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()
